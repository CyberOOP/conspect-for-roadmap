выполняется в конце кода

package main

import "fmt"

func main() {

fmt.Println("Начало main")

defer func() {

fmt.Println("Конец функции main")

}()

foo()
}

func foo() {

	defer func() {

		fmt.Println("DEFER 1") //положили самой первую

	}()

  

	defer func() {

		fmt.Println("DEFER 2") //сверху первой положили вторую

	}()

  

	defer func() {

		fmt.Println("DEFER 3") //сверху первой, второй положили сверху третью

	}()

  

	fmt.Println("FOO") // при вызове выведется foo

//потом поочереди берется из стопки

//самое верхнее было DEFER 3 поэтому выводится сначала DEFER 3

//потом DEFER 2, и в конце DEFER 1

//выводится в обратном порядке, тк берет их поочереди из stack-а

package main

//допустим есть переменная имеющая большое количество информации

//ее надо передать в какую либо функцию, при постоянном копировании переменной

//в другую функцию, занимается большое количество оперативной памяти, ну при большом объеме переменной

//для этого нужны указатели

//указатель позволяет иметь доступ к какому либо элементу память(переменной)

//но иногда нужно создать именно клон объекта, не изменяя изначальный

//указатель, это тоже тип переменной, те это тоже переменная

//пример: та же база данных, нужно не скопировать, а просто изменить

// какой либо объект, те иметь к нему доступ, через указатель

//но все указатели будут ссылаться УЖЕ НА ИЗМЕНЕННУЮ ПЕРЕМЕННУЮ

//но при передаче указателя мы его тоже копируем в функцию, только он будет занимать меньше места, чем если бы мы передавали какую то огромную переменную и тем самым скопировали бы ЕЁ

import "fmt"

  

func main() {

	number := 7 //переменная

	number1 := 5

	pointer1 := &number1 //присваиваем указателю переменную number1 через &

	fmt.Println(number1) //выведется сама переменная

	boo(pointer1) //передается указатель в функцию boo

	fmt.Println(number1) //оказывает влияние на исходную переменную, из 5 меняем на 10

	pointer := &number //указатель на переменную number

  

	fmt.Println(pointer) //└──╼ $go run указатели.go 0x37e7711ec110 выводит адрем переменной 

	fmt.Println(*pointer) //выводит содержимое переменной

	foo(pointer) //передали указатель pointer

}

  
func foo(n *int) { //функция ждет указатель переменной n с типом данных на который указатель указывает

	//* означает что тут будет указатель

	fmt.Println(n) //вывод адреса переменной по указателю

	fmt.Println(*n) //вывод содержимого переменной (значения)

}

  

func boo(n *int) { //переменная n принимает указатель с содержанием переменной int, с помощью звездочки передаем указаетль

	*n = 10 //через указатель мы меняем изначальную переменную
	чтобы поменять переменную, нужно добавить внчале звездочку
	*n

}

str := ""

for i := 1; i < len(fields); i++ { //перебор идет со второго слова, тк первое слово это команда (добавить или удалить) пошли с первого индекса и собрали слова в строчку

}
