файл создаем с .sh 'название.sh'
вводим #!/bin/bash и после снизу расписываем сам скрипт

==ПЕРЕМЕННЫЕ==
запись (перезапись) значениея
"имя"="значение" 
не ставить пробелы у равно
например: path=~/Docs

при вызове (чтении) переменной используем перед ней $
$path

#!/bin/bash
path=/home/kay/Downloads/file.txt
path2=$path/file.txt
echo "Path is $path2"
ВЫСВЕЧИВАЕТСЯ:
==Path is /home/kay/Downloads/file.txt ==

#!/bin/bash
path=/home/kay/Downloads/file.txt
path2=$path/file.txt
echo "Path is ${path}2"

вызовется сначала path и добавиться в конце двойка
ВЫСВЕЧИВАЕТСЯ:
Path is /home/kay/Downloads2

==АРГУМЕНТЫ==
передача аргументов скрипту
./scrip.sh arg1 arg2

при обработке внутри скрипта может отмечать аргументы как
$1 первый аргумент
$2 второй аргумент
$0 имя скрипта
$# количество аргументов (сколько передано аргументов всего (количество))

var="Первый аргумент равне $1"
echo "Всего было передано $# аргументов"

#!/bin/bash  
var1=$1  
var2=$2  
echo "Script is $0, arguments are $var1 and $var2 (total $#)"

ВЫСВЕЧИВАЕТСЯ ПРИ ВВОДЕ: ./she.sh go lang

Script is ./she.sh, arguments are go and lang (total 2)

КОД:
#!/bin/bash  
var1=$1  
var2=$2  
  
echo "Arguments are: \$1=$var1 \$2=$var2"

Выведет на экран запись в виде: ./script.sh go lang
Arguments are: $1=go $2=lang

для вывода конкретно текста ставим перед аргументами с $ ставим слэш \
типо \$1=var1 будет выведено $1=go

==ВЕТВЛЕНИЕ С УСЛОВИЯМИ==

поговорим про синтаксис if

if [[условие]]
then
	действие если условие истинно
fi

==УСЛОВИЯ (строки) ПРОВЕРКА:==
-z <строка>       строка пуста      проверить пустая ли строка
-n <строка>       строка не пуста     проверить не пустая ли строка

<стр1> == <стр2>     строки равны     проверить равны ли строки
<стр1> != <стр2>       строки не равны      проверить не равны ли строки

ПРИМЕР:
#!/bin/bash  
  
var1=$1  
var2=$2  
  
if [[ -z $1 ]]  
then  
   echo "Первый аргумент пустой!"  
fi  
  
if [[ $var1 == $var2 ]]  
then  
   echo "Переменные равны!"  
fi

ВЫСВЕЧИВАЕТСЯ ПРИ ВВОДЕ НИЧЕГО (ТЕ ОБЕ ПЕРЕМЕННЫЕ БУДУТ РАВНЫ):
./she.sh  
Первый аргумент пустой!  
Переменные равны!

==УСЛОВИЯ [числа (строки)]==
<число/строка> операция <число/строка>

-eq == равно
-ne (!=)
-lt (<)
-le меньше или равно
-gt (>) больше
-ge больше или равно

УСЛОВИЯ (файлы)
-e <путь> путь существует
-f <путь> это файл
-d <путь> это директория
-x <путь> размер файла больше 0
-x <путь> файл исполняемый

УСЛОВИЯ (логические)
! отрицание логического выражения
&& логическое "И"
|| логическое "ИЛИ"

==АЛЬТЕРНАТИВА IF, ЕСЛИ УСЛОВИЕ ИСТИННО ТО ВЫПОЛНИТЬ ОДНИ ДЕЙСТВИЯ, ЕСЛИ ЛОЖНО ТО ВЫПОЛНИТЬ ДРУГИЕ ДЕЙСТВИЯ:==

if [[условие]]
then
	условие истинно
else
	условие ложно
fi

НЕСКОЛЬКО ДЕЙСТВИЙ

if [[условие1]]
then
	действие при условии1 истинно
elif [[условие2]]
then
	действие если условие1 ложно
	но условие2 истинно
else
	действия если оба условия ложны
fi

ПРОВЕРКА ПЕРЕМЕННОЙ НА ЗНАЧЕНИЕ НЕСКОЛЬКО ЗНАЧЕНИЙ

case переменная in
знач1)
	действия если переменная = знач1
	;;
знач2)
	действия если переменная = знач2
	;;
звездочка) 
	действие если переменная
	или не равна одному из вариантов
esac
